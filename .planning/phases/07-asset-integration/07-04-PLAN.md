FUNCTION 1 — _load_preset_thumbnails() (called in _ready, after _preset_buttons are populated):
For each index i in range(_preset_buttons.size()):
  var thumb_path := "res://assets/textures/ui/avatar_presets/preset_%d.png" % (i + 1)
  var btn: Button = _preset_buttons[i]
  if ResourceLoader.exists(thumb_path, "Texture2D"):
    var tex: Texture2D = load(thumb_path) as Texture2D
    btn.icon = tex
    btn.expand_icon = true
    btn.text = ""
    btn.tooltip_text = tr("ui.avatar.preset_%d" % (i + 1))
  # If thumbnail not yet captured, button retains its text label (graceful degradation)

Place the _load_preset_thumbnails() call in _ready() AFTER the loop that wires _preset_buttons (after line ~235, after the for loop connecting preset button signals).

FUNCTION 2 — _generate_all_preset_thumbnails() (one-time developer capture tool):
This function is async (uses await) so must be declared: func _generate_all_preset_thumbnails() -> void
For each i in range(PRESETS.size()):
  _apply_config_to_preview(PRESETS[i])
  var sub_vp: SubViewport = _find_node("AvatarSubViewport") as SubViewport
  if sub_vp == null:
    push_warning("_generate_all_preset_thumbnails: AvatarSubViewport not found")
    return
  sub_vp.update_mode = SubViewport.UPDATE_ONCE
  RenderingServer.force_draw(false)
  await get_tree().process_frame   # wait for GPU→CPU transfer
  var img: Image = sub_vp.get_texture().get_image()
  if img == null or img.is_empty():
    push_warning("_generate_all_preset_thumbnails: blank image for preset %d" % i)
    continue
  var out_path := "res://assets/textures/ui/avatar_presets/preset_%d.png" % (i + 1)
  img.save_png(ProjectSettings.globalize_path(out_path))
  sub_vp.update_mode = SubViewport.UPDATE_WHEN_VISIBLE

After generation: push_print("All preset thumbnails generated. Commit assets/textures/ui/avatar_presets/ to git.")

Add a brief comment above the function: "# Developer-only: run this once in the editor to generate committed preset thumbnails."

Do NOT call _generate_all_preset_thumbnails() automatically from _ready() — it must only be triggered manually by the developer in a windowed Godot session.
