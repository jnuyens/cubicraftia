# Signal keepalive_timeout(peer_id: int)            emitted when miss count hits 6 before disconnect
# _simulate_peer_disconnect(peer_id: int) -> void  # test-only helper; triggers _on_host_disconnected if peer_id==1, else _disconnect_peer
#
# is_multiplayer_active() -> bool
# get_state() -> String  # one of the 10 state enum names
#
# Failover state machine transitions:
# CONNECTED_AS_PEER → _on_host_disconnected() → FAILOVER_DETECTING
# FAILOVER_DETECTING → _on_failover_timer_timeout() → FAILOVER_ELECTED (if am_i_elected)
# FAILOVER_ELECTED → _do_failover_elected() → FAILOVER_PROMOTING
# FAILOVER_PROMOTING → peer reconnects → FAILOVER_COMPLETE
# Note: _do_failover_elected() calls SessionRegistry.am_i_elected() and WorldSave.save_world_snapshot()
# Both are guarded with is_instance_valid(). In headless tests, these autoloads must be registered as
# Engine singletons before the state machine runs, or the transitions will silently no-op.
