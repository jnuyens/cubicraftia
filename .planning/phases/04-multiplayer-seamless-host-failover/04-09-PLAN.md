```gdscript
func _ready() -> void:
    if is_instance_valid(NetworkManager):
        NetworkManager.host_failover_started.connect(_on_failover_started)
        NetworkManager.host_failover_complete.connect(_on_failover_complete)

func _on_failover_started() -> void:
    visible = true
    _slow_subtitle_timer.start()  # 8-second one-shot → show slow subtitle

func _on_failover_complete() -> void:
    _slow_subtitle_timer.stop()
    # 0.22s fade-out then queue_free (matches panel animation duration)
    if _tween != null and _tween.is_running():
        _tween.kill()
    _tween = create_tween()
    _tween.tween_property(self, "modulate:a", 0.0, 0.22)\
        .set_ease(Tween.EASE_OUT).set_trans(Tween.TRANS_CUBIC)
    _tween.tween_callback(queue_free)
```
