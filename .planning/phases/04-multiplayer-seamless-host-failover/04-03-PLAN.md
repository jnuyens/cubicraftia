```gdscript
func _migrate_1_to_2() -> bool:
    push_warning("WorldSave._migrate_1_to_2: migrating world '%s' from schema v1 to v2." % world_id)
    if not _db.call("query", "BEGIN;"):
        push_error("WorldSave._migrate_1_to_2: could not BEGIN transaction.")
        return false
    var stmts := _v2_table_stmts()
    for i: int in stmts.size():
        if not _db.call("query", stmts[i]):
            _db.call("query", "ROLLBACK;")
            return false
    if not _db.call("query", "COMMIT;"):
        _db.call("query", "ROLLBACK;")
        return false
    return true
```

save_world_snapshot pattern (PATTERNS.md lines 286-308):
```gdscript
func save_world_snapshot(snapshot_id: String) -> bool:
    if _db == null:
        return false
    var inventory_blob := var_to_bytes(Inventory.get_all_state())
    _db.call("query_with_bindings",
        "INSERT OR REPLACE INTO snapshots(snapshot_id, created_at, chunk_blob, inventory_blob) VALUES (?, ?, ?, ?);",
        [snapshot_id, Time.get_unix_time_from_system(), PackedByteArray(), inventory_blob])
    return true
```

SessionRegistry election algorithm (PATTERNS.md lines 172-183):
```gdscript
func compute_elected_host() -> int:
    var candidates: Array = _surviving_peers.keys()
    candidates.sort_custom(func(a, b):
        var rtt_a: float = _rtt_rolling_avg.get(a, INF)
        var rtt_b: float = _rtt_rolling_avg.get(b, INF)
        if absf(rtt_a - rtt_b) < 1.0:
            return _join_order.get(a, INF) < _join_order.get(b, INF)
        return rtt_a < rtt_b
    )
    return candidates[0] if candidates.size() > 0 else multiplayer.get_unique_id()

func am_i_elected() -> bool:
    return compute_elected_host() == multiplayer.get_unique_id()
```

RTT rolling average update (from 04-RESEARCH.md keepalive pattern):
```gdscript
func update_peer_rtt(peer_id: int, rtt_ms: float) -> void:
    # 30-second rolling average approximated via EWMA (alpha=0.1 for ~30s window at 1Hz)
    var current: float = _rtt_rolling_avg.get(peer_id, rtt_ms)
    _rtt_rolling_avg[peer_id] = current * 0.9 + rtt_ms * 0.1
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| WorldSave snapshot write | Only host writes snapshots; peers receive via RPC snapshot reset |
| SessionRegistry election | Computed locally from shared data; no signaling round-trip needed |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-03-T | Tampering | WorldSave snapshot overwrite | mitigate | INSERT OR REPLACE with parameterised query; no string concatenation in query_with_bindings calls |
| T-04-03-T2 | Tampering | Election result forgery | accept | Election is local + deterministic from shared data; no peer can inject a different result because every peer computes independently; net result: honest peers always converge |
| T-04-03-I | Information Disclosure | snapshot blob contents | accept | snapshot is stored in user://worlds/ which is sandboxed per platform; world data is not sensitive PII |
</threat_model>
