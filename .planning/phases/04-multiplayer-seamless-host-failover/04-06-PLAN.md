```gdscript
func _setup_panel_style() -> void:
    var style := StyleBoxFlat.new()
    style.bg_color = COLOR_NAVY
    if layout == "sidebar":
        style.corner_radius_top_left = 16
        style.corner_radius_bottom_left = 16
        # right corners = 0
    else:
        style.corner_radius_top_left = 16
        style.corner_radius_top_right = 16
        # bottom corners = 0
    _body.add_theme_stylebox_override("panel", style)

func _animate_to(target_y: float) -> void:
    if _tween != null and _tween.is_running():
        _tween.kill()
    _tween = create_tween()
    _tween.tween_property(self, "position:y", target_y, TWEEN_DURATION_S)\
        .set_ease(Tween.EASE_OUT).set_trans(Tween.TRANS_CUBIC)

func _snap_to_nearest() -> void:
    var mid: float = (_expanded_y + _collapsed_y) * 0.5
    if position.y <= mid:
        _animate_to(_expanded_y); _is_expanded = true; _notify_mobile_overlay(true)
    else:
        _animate_to(_collapsed_y); _is_expanded = false; _notify_mobile_overlay(false)
```

| Boundary | Description |
|----------|-------------|
| Sign-in form | User-supplied email/password sent to Supabase via HTTPS; never logged |
| Friends panel | Displays data from FriendsClient (Supabase JWT-authenticated); no raw friend UID displayed in UI |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-06-I | Information Disclosure | Error messages revealing server details | mitigate | Error labels show only tr() user-facing strings; HTTP codes and server messages are never surfaced to the user |
| T-04-06-T | Tampering | OAuth stub accepts any provider | accept | SIWA/Google sign-in stubs emit sign_in_failed until Phase 5/6; no false authentication possible |
| T-04-06-S | Spoofing | Friend search result tampering | accept | Friend search results come from Supabase via JWT-authenticated call; RLS profiles policy allows all authenticated reads (intended for friend search) |
</threat_model>
