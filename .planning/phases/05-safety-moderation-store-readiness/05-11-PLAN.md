    ```yaml
    export_ios:
      name: "Export iOS (TestFlight)"
      runs-on: macos-15
      if: ${{ secrets.APPLE_CERTIFICATE_P12 != '' }}
      needs: [scope_checks]
      steps:
        - uses: actions/checkout@v4
        - name: Setup Godot
          # Use the same Godot setup step as other export jobs (download Godot 4.6 binary)
        - name: Import iOS export templates
          # Download Godot 4.6 iOS export templates
        - name: Setup Apple signing
          env:
            APPLE_CERTIFICATE_P12: ${{ secrets.APPLE_CERTIFICATE_P12 }}
            APPLE_CERTIFICATE_PASSWORD: ${{ secrets.APPLE_CERTIFICATE_PASSWORD }}
            APPLE_PROVISIONING_PROFILE: ${{ secrets.APPLE_PROVISIONING_PROFILE }}
          run: |
            # Decode and install certificate + provisioning profile
            echo "$APPLE_CERTIFICATE_P12" | base64 --decode > certificate.p12
            security create-keychain -p "" build.keychain
            security import certificate.p12 -k build.keychain -P "$APPLE_CERTIFICATE_PASSWORD" -A
            security set-key-partition-list -S apple-tool:,apple: -s -k "" build.keychain
            mkdir -p ~/Library/MobileDevice/Provisioning\ Profiles
            echo "$APPLE_PROVISIONING_PROFILE" | base64 --decode > ~/Library/MobileDevice/Provisioning\ Profiles/cubicraftia.mobileprovision
        - name: Export iOS IPA
          run: |
            godot --headless --export-release "iOS" build/Cubicraftia.ipa
        - uses: dulvui/godot-ios-upload@v4
          with:
            api-key-id: ${{ secrets.APP_STORE_CONNECT_KEY_ID }}
            api-key-issuer-id: ${{ secrets.APP_STORE_CONNECT_ISSUER_ID }}
            api-private-key: ${{ secrets.APP_STORE_CONNECT_PRIVATE_KEY }}
            ipa-path: build/Cubicraftia.ipa
    ```
