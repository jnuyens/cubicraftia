    - `var _target_world_pos: Vector3 = Vector3.ZERO`
    - `var _has_target: bool = false`

    1. _connect_ftue_signals()
    2. _activate_step(1) — show step 1 narration, enable chest highlight

    - Update _progress_bar.value = step
    - Update _progress_label.text = tr("ui.ftue.progress_label").format({"n": step, "total": STEP_COUNT})
    - Step 1: set _target_world_pos to the starter chest position; _has_target = true; enable ChestEntity.ftue_highlight uniform
    - Step 2: _find_nearest_tree() via call_deferred (avoids hitch per UI-SPEC Recommendation 3); _has_target = true; start GPUParticles3D highlight for 10s
    - Step 3: _target_world_pos = Vector3.ZERO; _has_target = false; BrickPalette._set_ftue_highlight("wood_plank", true)
    - Step 4: _has_target = false; BrickPalette._set_ftue_highlight("wood_plank", false); _show_completion()

    `func _show_completion() -> void:`:
    - Animate narration label alpha 0→1 over 0.5s
    - OnboardingTelemetry.log("ftue_complete")
    - WorldSave.set_world_meta("ftue_complete", var_to_bytes(true))
    - Start a 3.0s Timer: on timeout → fade out entire overlay (Tween modulate.a 1→0, 0.5s) → queue_free()

    `func _advance_step() -> void:`:
    - _current_step += 1
    - OnboardingTelemetry.log("ftue_step_%d_complete" % (_current_step - 1))
    - if _current_step > STEP_COUNT: _show_completion(); return
    - _activate_step(_current_step)

    `func _on_chest_opened() -> void:`: if _current_step == 1: _advance_step()
    `func _on_item_added(builder_id: String, def_id: String, _count: int) -> void:`: if _current_step == 2 and def_id == "wood_log": _advance_step()
    `func _on_stud_placed(anchor_cell: Vector3i, definition: BrickDefinition, colour_index: int, rotation: int) -> void:`: if _current_step == 3 and definition != null and definition.brick_id == "wood_plank": _advance_step()

    `func _process(delta: float) -> void:`:
    - _update_arrow_direction() if _has_target
    - Check WorldClock for DUSK transition: if WorldClock time_of_day is DUSK and _current_step < 4 and not _night_hint_shown: _show_night_hint()

    `func _update_arrow_direction() -> void:`: per 06-RESEARCH.md Pattern code — unproject target pos, clamp to screen edge, set arrow position and rotation, arrow.visible = not on-screen.

    `func _find_nearest_tree() -> void:` (deferred call): VoxelTool scan in 20-block radius from spawn, find wood_log voxel, set _target_world_pos, _has_target = true.

    `func _show_night_hint() -> void:`: fade in _night_hint_label (brick-white 70% alpha), set text tr("ui.ftue.night_hint"). _night_hint_shown = true. Does not block step 4.

    - AudioStreamPlayer for step-complete chime (null stream, no crash — placeholder for future SFX)
